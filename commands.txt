# Termux XFCE Desktop - ONE BY ONE
# Paste these in Termux one at a time, wait for each to finish.

# 1. Update packages
pkg update -y && pkg upgrade -y

# 2. Unlock GUI packages
pkg install x11-repo -y

# 3. Install desktop + VNC + dbus
pkg install xfce4 tigervnc dbus -y

# 4. Set VNC password (max 8 chars, remember it)
vncpasswd

# 5. Create the startup file
mkdir -p ~/.vnc
printf '#!/data/data/com.termux/files/usr/bin/sh\nexec dbus-launch --exit-with-session xfce4-session\n' > ~/.vnc/xstartup
chmod +x ~/.vnc/xstartup

# 6. Start the desktop (this makes it visible to your PC)
vncserver -localhost no :1

# 7. Find your phone IP
ifconfig wlan0 | grep inet

# Then on your PC: open TightVNC Viewer, connect to PHONE-IP:5901
# Stop:  vncserver -kill :1
# Start: vncserver -localhost no :1
